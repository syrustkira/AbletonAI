cmake_minimum_required(VERSION 3.16)
project(n0te_native C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(n0te_dsp STATIC n0te_dsp.c n0te_ring.c)
target_include_directories(n0te_dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(n0te_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(n0te_native_tests test_native.c)
target_link_libraries(n0te_native_tests PRIVATE n0te_dsp m)
target_compile_options(n0te_native_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME n0te_native_tests COMMAND n0te_native_tests)
